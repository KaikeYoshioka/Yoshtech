Produtos por categoria

Rota: GET /produtos/consultas/por-categoria

SELECT categoria, COUNT(id), SUM(quantidadeEstoque), AVG(preco), MIN(preco), MAX(preco),
       SUM(preco * quantidadeEstoque) AS valorEmEstoque
FROM Produtos GROUP BY categoria ORDER BY valorEmEstoque DESC;
